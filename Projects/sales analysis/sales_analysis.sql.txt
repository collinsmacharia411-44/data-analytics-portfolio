USE sales_analysis;


-- ============================================================
-- 1. REVENUE BY PRODUCT
-- Business question:
-- Which products generate the most revenue?
-- ============================================================

SELECT
    product,
    SUM(units) AS total_units_sold,
    SUM(units * unit_price) AS total_revenue
FROM sales
GROUP BY product
ORDER BY total_revenue DESC;


-- ============================================================
-- 2. SALES PERFORMANCE BY REGION
-- Business question:
-- Which regions generate the most revenue?
-- ============================================================

SELECT
    region,
    SUM(units) AS total_units_sold,
    SUM(units * unit_price) AS total_revenue
FROM sales
GROUP BY region
ORDER BY total_revenue DESC;


-- ============================================================
-- 3. REVENUE BY CATEGORY
-- Business question:
-- Which product categories contribute the most revenue?
-- ============================================================

SELECT
    category,
    SUM(units) AS total_units_sold,
    SUM(units * unit_price) AS total_revenue,
    ROUND(
        SUM(units * unit_price) * 100.0 /
        (SELECT SUM(units * unit_price) FROM sales),
        2
    ) AS revenue_percentage
FROM sales
GROUP BY category
ORDER BY total_revenue DESC;


-- ============================================================
-- 4. OVERALL BUSINESS KPIs
-- Business question:
-- What is the overall sales performance?
-- ============================================================

SELECT
    SUM(units * unit_price) AS total_revenue,
    SUM(units) AS total_units_sold,
    COUNT(*) AS total_sales_records,
    ROUND(AVG(units * unit_price), 2) AS average_revenue_per_record
FROM sales;


-- ============================================================
-- 5. TOP PRODUCT IN EACH REGION
-- Business question:
-- What is the highest-revenue product in each region?
--
-- CTEs are used to calculate revenue by region and product.
-- DENSE_RANK() then ranks products within each region.
-- ============================================================

WITH product_region_sales AS (

    SELECT
        region,
        product,
        SUM(units * unit_price) AS revenue
    FROM sales
    GROUP BY region, product

),

ranked_products AS (

    SELECT
        region,
        product,
        revenue,
        DENSE_RANK() OVER (
            PARTITION BY region
            ORDER BY revenue DESC
        ) AS product_rank
    FROM product_region_sales

)

SELECT
    region,
    product,
    revenue
FROM ranked_products
WHERE product_rank = 1
ORDER BY region;


-- ============================================================
-- 6. REVENUE CONTRIBUTION BY PRODUCT
-- Business question:
-- What percentage of total revenue does each product generate?
-- ============================================================

SELECT
    product,
    SUM(units) AS total_units_sold,
    SUM(units * unit_price) AS total_revenue,
    ROUND(
        SUM(units * unit_price) * 100.0 /
        (SELECT SUM(units * unit_price) FROM sales),
        2
    ) AS revenue_percentage
FROM sales
GROUP BY product
ORDER BY total_revenue DESC;


-- ============================================================
-- 7. REVENUE CONTRIBUTION BY REGION
-- Business question:
-- What percentage of total revenue does each region generate?
-- ============================================================

SELECT
    region,
    SUM(units) AS total_units_sold,
    SUM(units * unit_price) AS total_revenue,
    ROUND(
        SUM(units * unit_price) * 100.0 /
        (SELECT SUM(units * unit_price) FROM sales),
        2
    ) AS revenue_percentage
FROM sales
GROUP BY region
ORDER BY total_revenue DESC;


-- ============================================================
-- 8. PRODUCT AND REGION PERFORMANCE
-- Business question:
-- How does each product perform across different regions?
-- ============================================================

SELECT
    region,
    product,
    SUM(units) AS total_units_sold,
    SUM(units * unit_price) AS total_revenue
FROM sales
GROUP BY region, product
ORDER BY region, total_revenue DESC;


-- ============================================================
-- 9. CHECK FOR MISSING VALUES
-- Business question:
-- Are there missing values in important sales fields?
-- ============================================================

SELECT
    SUM(CASE WHEN order_id IS NULL THEN 1 ELSE 0 END)
        AS missing_order_ids,

    SUM(CASE WHEN product IS NULL THEN 1 ELSE 0 END)
        AS missing_products,

    SUM(CASE WHEN category IS NULL THEN 1 ELSE 0 END)
        AS missing_categories,

    SUM(CASE WHEN region IS NULL THEN 1 ELSE 0 END)
        AS missing_regions,

    SUM(CASE WHEN units IS NULL THEN 1 ELSE 0 END)
        AS missing_units,

    SUM(CASE WHEN unit_price IS NULL THEN 1 ELSE 0 END)
        AS missing_prices
FROM sales;


-- ============================================================
-- 10. CHECK FOR INVALID VALUES
-- Business question:
-- Are there negative quantities or prices?
-- ============================================================

SELECT *
FROM sales
WHERE units < 0
   OR unit_price < 0;
